作业要求：
采用LNMP架构，通过httpd实现虚拟主机，通过dns做域名解析
要求客户端访问discuz.lab0.com访问的是discuz论坛网站
要求客户端访问wordpress.lab0.com访问的是wordpress博客网站
wordpress采用4.5版本，discuz采用x3.5版本

## 安装软件包
```bash
[root@dpeak ~]# yum install -y nginx mysql php bind
```

## 配置dns
### 修改主配置文件
```bash
[root@dpeak ~]# vim /etc/named.conf
options {
        listen-on port 53 { any; };
        listen-on-v6 port 53 { ::1; };
        directory       "/var/named";
        dump-file       "/var/named/data/cache_dump.db";
        statistics-file "/var/named/data/named_stats.txt";
        memstatistics-file "/var/named/data/named_mem_stats.txt";
        secroots-file   "/var/named/data/named.secroots";
        recursing-file  "/var/named/data/named.recursing";
        allow-query     { any; };



```
### 修改区域配置文件
```bash
[root@dpeak ~]# vim /etc/named.rfc1912.zones
zone "lab0.com" IN {
        type master;
        file "lab0.com";
        allow-update { none; };
};

zone "200.168.192.in-addr.arpa" IN {
        type master;
        file "200.168.192.in-addr.arpa";
        allow-update { none; };
};
```
### 创建正向解析文件
```bash
[root@dpeak ~]# cd /var/named/
[root@dpeak named]# ls
data  dynamic  named.ca  named.empty  named.localhost  named.loopback  slaves
[root@dpeak named]# cp -a named.localhost lab0.com
[root@dpeak named]# vim lab0.com
$TTL 1D
@       IN SOA  dns.lab0.com. admin.lab0.com. (
                                        0       ; serial
                                        1D      ; refresh
                                        1H      ; retry
                                        1W      ; expire
                                        3H )    ; minimum
        NS      dns.lab0.com.
        A       127.0.0.1
        AAAA    ::1
dns     A       192.168.200.128
discuz  A       192.168.200.128
wordpress       A       192.168.200.128

```
### 创建反向解析文件
```bash
[root@dpeak named]# cp -a named.loopback 200.168.192.in-addr.arpa
[root@dpeak named]# vim 200.168.192.in-addr.arpa
$TTL 1D
@       IN SOA  dns.lab0.com. admin.lab0.com. (
                                        0       ; serial
                                        1D      ; refresh
                                        1H      ; retry
                                        1W      ; expire
                                        3H )    ; minimum
        NS      dns.lab0.com.
        A       127.0.0.1
        AAAA    ::1
128     PTR     discuz.lab0.com.
128     PTR     wordpress.lab0.com.

```
