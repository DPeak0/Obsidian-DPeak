要求：每一个步骤都要使用命令和截图，遇到了任何报错信息 也要进行截图，并且同时说明报错的问题以及如何解决的，将其整理到一个word文件里面私发俺

## 准备工作
- 克隆3台虚拟机，固件类型为BIOS
- 关闭VMware的DHCP服务器

## 配置PXE服务端
### 一、配置本地YUM仓库
```bash
[root@dpeak ~]# cd /etc/yum.repos.d/
[root@dpeak yum.repos.d]# ls
Rocky-AppStream.repo  Rocky-HighAvailability.repo  Rocky-ResilientStorage.repo
Rocky-BaseOS.repo     Rocky-Media.repo             Rocky-RT.repo
Rocky-Devel.repo      Rocky-Plus.repo              Rocky-Sources.repo
Rocky-Extras.repo     Rocky-PowerTools.repo
[root@dpeak yum.repos.d]# mkdir bak
[root@dpeak yum.repos.d]# mv *.repo bak/
[root@dpeak yum.repos.d]# cat > dvd.repo <<EOF
> [BaseOS]
> name=BaseOS
> baseurl=file:///media/BaseOS
> gpgcheck=0
>
> [AppStream]
> name=AppStream
> baseurl=file:///media/AppStream
> gpgcheck=0
> EOF
[root@dpeak yum.repos.d]# mount /dev/sr0 /media/

[root@dpeak yum.repos.d]# yum clean all && yum makecache

```
### 二、安装所需软件包
```bash
[root@dpeak yum.repos.d]# yum install -y httpd tftp-server.x86_64 dhcp-server.x86_64 syslinux-nonlinux.noarch
```
### 三、关闭防火墙和SELINUX
```bash
[root@dpeak yum.repos.d]# systemctl stop firewalld.service
[root@dpeak yum.repos.d]# setenforce 0
```
### 四、配置DHCP服务端
```bash
[root@dpeak ~]# cat /etc/dhcp/dhcpd.conf
#
# DHCP Server Configuration file.
#   see /usr/share/doc/dhcp-server/dhcpd.conf.example
#   see dhcpd.conf(5) man page
#

[root@dpeak ~]# cat /usr/share/doc/dhcp-server/dhcpd.conf.example >> /etc/dhcp/dhcpd.conf
[root@dpeak ~]# vim /etc/dhcp/dhcpd.conf
subnet 192.168.200.0 netmask 255.255.255.0 {
  range 192.168.200.100 192.168.200.200;
  option domain-name-servers 223.5.5.5;
  option domain-name "internal.example.org";
  option routers 192.168.200.2;
  option broadcast-address 192.168.200.255;
  default-lease-time 600;
  max-lease-time 7200;
  next-server 192.168.200.128;
  filename "pxelinux.0";
}

```
### 五、配置TFTP
```bash
[root@dpeak ~]# cp /media/isolinux/* /var/lib/tftpboot/
[root@dpeak ~]# ls /var/lib/tftpboot/
boot.cat  grub.conf   isolinux.bin  ldlinux.c32   libutil.c32  splash.png  vesamenu.c32
boot.msg  initrd.img  isolinux.cfg  libcom32.c32  memtest      TRANS.TBL   vmlinuz

[root@dpeak ~]# cp /usr/share/syslinux/pxelinux.0 /var/lib/tftpboot/
[root@dpeak ~]# ls /var/lib/tftpboot/
boot.cat   initrd.img    ldlinux.c32   memtest     TRANS.TBL
boot.msg   isolinux.bin  libcom32.c32  pxelinux.0  vesamenu.c32
grub.conf  isolinux.cfg  libutil.c32   splash.png  vmlinuz

[root@dpeak ~]# mkdir /var/lib/tftpboot/pxelinux.cfg
[root@dpeak ~]# cp /var/lib/tftpboot/isolinux.cfg /var/lib/tftpboot/pxelinux.cfg/default
```
