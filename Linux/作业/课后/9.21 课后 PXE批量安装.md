晚自习：准备3台虚拟机，1台作为pxe服务端，通过pxe+kickstart实现2台主机的自动化部署，要求安装好之后存在一个用户叫做devops，并且密码是1，同时还在/opt目录下创建一个名为devops的文件（可以借助ks脚本文件中的%post%内容，也就是安装好之后要执行什么命令实现）
要求：每一个步骤都要使用命令和截图，遇到了任何报错信息 也要进行截图，并且同时说明报错的问题以及如何解决的，将其整理到一个word文件里面私发俺

## 准备工作
- 准备3台虚拟机，固件类型为BIOS
- 关闭VMware的DHCP服务器

## 配置PXE服务端

### 二、安装所需软件包
```bash
[root@dpeak yum.repos.d]# yum install -y httpd tftp-server.x86_64 dhcp-server.x86_64 syslinux-nonlinux.noarch
```
### 三、关闭防火墙和SELINUX
```bash
[root@dpeak yum.repos.d]# systemctl stop firewalld.service
[root@dpeak yum.repos.d]# setenforce 0
```
### 四、配置DHCP服务端
```bash
[root@dpeak ~]# cat /etc/dhcp/dhcpd.conf
#
# DHCP Server Configuration file.
#   see /usr/share/doc/dhcp-server/dhcpd.conf.example
#   see dhcpd.conf(5) man page
#

[root@dpeak ~]# cat /usr/share/doc/dhcp-server/dhcpd.conf.example >> /etc/dhcp/dhcpd.conf
[root@dpeak ~]# vim /etc/dhcp/dhcpd.conf
subnet 192.168.200.0 netmask 255.255.255.0 {
  range 192.168.200.100 192.168.200.200;
  option domain-name-servers 223.5.5.5;
  option domain-name "internal.example.org";
  option routers 192.168.200.2;
  option broadcast-address 192.168.200.255;
  default-lease-time 600;
  max-lease-time 7200;
  next-server 192.168.200.128;
  filename "pxelinux.0";
}

```
### 五、配置TFTP
```bash
#将引导文件复制到tftp目录下
[root@dpeak ~]# cp /media/isolinux/* /var/lib/tftpboot/
[root@dpeak ~]# ls /var/lib/tftpboot/

boot.msg  initrd.img  isolinux.cfg  libcom32.c32  memtest      TRANS.TBL   vmlinuz

#将引导程序文件复制到tftp目录下
[root@dpeak ~]# cp /usr/share/syslinux/pxelinux.0 /var/lib/tftpboot/
[root@dpeak ~]# ls /var/lib/tftpboot/
boot.cat   initrd.img    ldlinux.c32   memtest     TRANS.TBL

#引导配置文件必须是存放到pxelinux.cfg/default中

[root@dpeak ~]# cp /var/lib/tftpboot/isolinux.cfg /var/lib/tftpboot/pxelinux.cfg/default
```

### 七、配置kickstart脚本
```bash
[root@dpeak ~]# mkdir /var/www/html/ks
[root@dpeak ~]# vim /var/www/html/ks/ks.cfg
[root@dpeak ~]# cat /var/www/html/ks/ks.cfg
#platform=x86, AMD64, 或 Intel EM64T
#version=DEVEL
# Install OS instead of upgrade
install
# Keyboard layouts
keyboard 'us'
# Root password
rootpw --iscrypted $1$lOboUwAI$y19Mo0jcA7ss1AXi.xkFC1
# System language
lang en_US
# System authorization information
auth  --useshadow  --passalgo=sha512
# Use graphical install
graphical
# SELinux configuration
selinux --disabled
# Do not configure the X Window System
skipx


# Firewall configuration
firewall --disabled
# Network information
network  --bootproto=dhcp --device=ens160
# Reboot after installation
reboot
# System timezone
timezone Asia/Shanghai
# Use network installation
url --url="http://192.168.200.128/iso"
# System bootloader configuration
bootloader --location=mbr
# Clear the Master Boot Record
zerombr
# Partition clearing information
clearpart --all --initlabel
# Disk partitioning information
part /boot --fstype="xfs" --size=512
part / --fstype="xfs" --size=10240

%post
useradd devops
echo 1 | passwd --stdin devops
touch /opt/devops
%end

%packages
@fonts

%end

```
