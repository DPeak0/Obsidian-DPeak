准备一台宿主机
配置8U 8G内存
需求如下：
配置一个桥接网络br0
安装KVM虚拟化套件
创建一台rockylinux8系统的虚拟机
要求配置为1CPU 1G内存 20G精简模式磁盘
网络选择br0桥接网络
选择最小化安装
最终可以通过virsh console 登录到这个虚拟机中
```bash
[root@dpeak ~]# nmcli connection delete ens160
[root@dpeak ~]# nmcli connection add type bridge ifname br0 con-name br0
Connection 'br0' (f4d44756-4404-41a7-8ab2-004bacaa245a) successfully added.

[root@dpeak ~]# nmcli connection add type bridge-slave ifname ens160 con-name ens160 master br0 
Connection 'ens160' (149864d4-3cc6-49b6-8280-972a5f23eacf) successfully added.

```

[root@dpeak ~]# yum groupinstall 'Virtualization Host' -y
