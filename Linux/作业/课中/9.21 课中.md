# PXE+kickstart实战案例

准备2台主机：
1台pxe服务端
1台pxe客户端
要求通过pxe+kickstart方式来实现pxe客户端系统自动安装

### 配置本地yum仓库
```bash
[root@dpeak ~]# cd /etc/yum.repos.d/
[root@dpeak yum.repos.d]# mkdir bak
[root@dpeak yum.repos.d]# rm -rf *.repo
[root@dpeak yum.repos.d]# cat > dvd.repo <<EOF
> [BaseOS]
> name=BaseOS
> baseurl=file:///media/BaseOS
> gpgcheck=0
>
> [AppStream]
> name=AppStream
> baseurl=file:///media/AppStream
> gpgcheck=0
> EOF
[root@dpeak yum.repos.d]# mount /dev/sr0 /media/
mount: /media: WARNING: device write-protected, mounted read-only.
[root@dpeak yum.repos.d]# yum clean all
13 files removed
[root@dpeak yum.repos.d]# yum makecache
BaseOS                                                3.9 MB/s | 2.5 MB     00:00
AppStream                                             4.3 MB/s | 7.1 MB     00:01
Last metadata expiration check: 0:00:01 ago on Mon 21 Sep 2026 11:24:40 AM CST.
Metadata cache created.

```

### 安装所需软件包
```bash
[root@dpeak yum.repos.d]# yum install -y httpd dhcp-server tftp-server syslinux-nonlinux

```
### 关闭服务器和selinux
```bash
[root@dpeak yum.repos.d]# systemctl stop firewalld.service
[root@dpeak yum.repos.d]# setenforce 0
```

### 配置DHCP服务器
```bash
[root@dpeak yum.repos.d]# cd /etc/dhcp/
[root@dpeak dhcp]# ls
dhclient.conf  dhclient.d  dhcpd6.conf  dhcpd.conf
[root@dpeak dhcp]# cat dhcpd.conf
#
# DHCP Server Configuration file.
#   see /usr/share/doc/dhcp-server/dhcpd.conf.example
#   see dhcpd.conf(5) man page
#
[root@dpeak dhcp]# cat /usr/share/doc/dhcp-server/dhcpd.conf.example > dhcpd.conf
[root@dpeak dhcp]# vim dhcpd.conf
[root@dpeak dhcp]# cat dhcpd.conf

# A slightly different configuration for an internal subnet.
subnet 192.168.200.0 netmask 255.255.255.0 {
  range 192.168.200.100 192.168.200.200;
  option domain-name-servers 223.5.5.5;
  option domain-name "internal.example.org";
  option routers 192.168.200.2;
  option broadcast-address 192.168.200.255;
  default-lease-time 600;
  max-lease-time 7200;
  next-server 192.168.200.128;
  filename "pxelinux.0";
}



```
### 配置tftp
```bash
[root@dpeak dhcp]# cd /var/lib/tftpboot/
[root@dpeak tftpboot]# cp /media/isolinux/* .
[root@dpeak tftpboot]# cp /usr/share/syslinux/pxelinux.0 .
[root@dpeak tftpboot]# mkdir pexlinux.cfg
[root@dpeak tftpboot]# cp isolinux.cfg pexlinux.cfg/default

```
### 配置httpd
```bash
[root@dpeak tftpboot]# cd /var/www/html/
[root@dpeak html]# mkdir iso
[root@dpeak html]# mount /dev/sr0 iso/
mount: /var/www/html/iso: WARNING: device write-protected, mounted read-only.

[root@dpeak html]# ls iso/
AppStream  BaseOS  EFI  images  isolinux  LICENSE  media.repo  TRANS.TBL

```
