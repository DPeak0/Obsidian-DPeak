
文件系统永久挂载和修复实战
准备工作：给你自己的Linux虚拟机额外添加一个块SCSI的20GB 磁盘，要求直接对整块磁盘进行格式化，文件系统类型
为 xfs，要求开机自动挂载到/data目录
上一个要求做完之后，请你在/etc/fstab文件中添加以下内容
```bash
/dev/sda /media xfs default 0 0
```
请你重启系统，进行文件系统故障修复，要求修复之后可以正常进入操作系统
```bash
[root@app-servser-01 ~]# cat /etc/fstab | grep sdb
/dev/sdb /data xfs defaults 0 0


[root@app-servser-01 ~]# df | grep sdb
/dev/sdb        20961280  179192  20782088   1% /data

#可以正常挂载，接下来模拟故障，故意写错defaults
[root@app-servser-01 ~]# cat /etc/fstab | grep sdb
/dev/sdb /data xfs default 0 0
[root@app-servser-01 ~]# mount -a
mount: /data: wrong fs type, bad option, bad superblock on /dev/sdb, missing codepage or helper program, or other error.


[root@app-servser-01 ~]# reboot

```

。创建一个1G大小的文件/swap-file，要求将其格式化为 swap 文件系统，将其永久挂载
```bash
[root@app-servser-01 ~]# dd if=/dev/zero of=/swap-file bs=1M count=1024
1024+0 records in
1024+0 records out
1073741824 bytes (1.1 GB, 1.0 GiB) copied, 5.62807 s, 191 MB/s
[root@app-servser-01 ~]# chmod 600 /swap-file
[root@app-servser-01 ~]# swapon /swap-file
swapon: /swap-file: swapon failed: Device or resource busy
[root@app-servser-01 ~]# swapon -s
Filename                                Type            Size    Used    Priority
/dev/sdb1                               partition       1048572 0       -2
/swap-file                              file            1048572 0       -3

[root@app-servser-01 ~]# cat /etc/fstab  | grep  swap-file
/swap-file none swap defaults  0 0

[root@app-servser-01 ~]# swapon -a
[root@app-servser-01 ~]# swapon -s
Filename                                Type            Size    Used    Priority
/dev/sdb1                               partition       1048572 240     -2
/swap-file                              file            1048572 0       -3

```
