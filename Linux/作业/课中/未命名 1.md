# apache

准备2台rockylinux8虚拟机
一台充当web主机和dns服务器
一台充当客户端
一个是DNS的正向解析配置
一个是apache的基于域名的虚拟主机配置
要求客户端的dns地址指向dns服务器
  基于域名的虚拟主机
访问apache01.lab0.cn的时候，网站内容显示的是apache01
访问apache02.lab0.cn的时候，网站内容显示的是apache02
最后的验证截图

## 配置DNS
```bash

zone "lab0.cn" IN {
        type master;
        file "lab0.cn";
        allow-update { none; };
};

zone "200.168.192.in-addr.arpa" IN {
        type master;
        file "200.168.192.in-addr.arpa";
        allow-update { none; };
};


```

```bash
[root@dpeak named]# cat /etc/named.rfc1912.zones
zone "lab0.cn" IN {
        type master;
        file "lab0.cn";
        allow-update { none; };
};

zone "200.168.192.in-addr.arpa" IN {
        type master;
        file "200.168.192.in-addr.arpa";
        allow-update { none; };
};


```

```bash
[root@dpeak named]# cd /var/named/
[root@dpeak named]# cp -a named.localhost lab0.cn
[root@dpeak named]# vim lab0.cn
$TTL 1D
@       IN SOA  dns.lab0.cn. admin.lab0.cn. (
                                        0       ; serial
                                        1D      ; refresh
                                        1H      ; retry
                                        1W      ; expire
                                        3H )    ; minimum
        NS      dns.lab0.cn.
        A       127.0.0.1
        AAAA    ::1
dns     A       192.168.200.128
apache01        A       192.168.200.128
apache02        A       192.168.200.128

```

# nginx

```bash

[root@dpeak ~]# vim /etc/nginx/nginx.conf

    upstream webservers {
        server 192.168.200.129;
        server 192.168.200.130;
    }

    server {
        listen       80 default_server;
        listen       [::]:80 default_server;
        server_name  _;
        root         /usr/share/nginx/html;

        # Load configuration files for the default server block.
        include /etc/nginx/default.d/*.conf;

        location / {
                proxy_pass http://webservers;
        }

        error_page 404 /404.html;
            location = /40x.html {
        }

        error_page 500 502 503 504 /50x.html;
            location = /50x.html {
        }
    }
```
