## 编辑Redis配置文件
### node1
```bash
[root@node1 ~]# mkdir -p /etc/redis /var/log/redis /var/lib/redis
[root@node1 ~]# vim /etc/redis/6379.conf
[root@node1 ~]# cat /etc/redis/6379.conf
bind 0.0.0.0
port 6379
protected-mode no
daemonize yes
dir /var/lib/redis
logfile "/var/log/redis/redis.log"
requirepass Redis@123
masterauth Redis@123
dbfilename dump.rdb
```

## 配置守护进程
```Bash
[root@node1 ~]# cat > /etc/systemd/system/redis.service << 'EOF'
> [Unit]
> Description=Redis In-Memory Data Store
> After=network.target
>
> [Service]
> Type=forking
> ExecStart=/usr/local/redis/bin/redis-server /etc/redis/6379.conf
> ExecStop=/usr/local/redis/bin/redis-cli -h 127.0.0.1 -p 6379 -a "Redis@123" shutdown
> Restart=always
>
> [Install]
> WantedBy=multi-user.target
> EOF

[root@node3 redis-7.2.4]# systemctl daemon-reload
[root@node3 redis-7.2.4]# systemctl enable --now redis
```


# Redis哨兵模式
### node1、node2、node3上配置
```bash
[root@node1 ~]# cat << 'EOF' > /etc/redis/redis-sentinel.conf
> # 允许外部网络访问 Sentinel 端口（默认 26379）
> bind 0.0.0.0
> port 26379
> daemonize no
> protected-mode no
> pidfile /var/run/redis-sentinel.pid
> logfile /var/log/redis/sentinel.log
> dir /tmp
>
> # 核心：监控的主节点名字（mymaster）、IP（node1 的私网 IP）、端口，以及触发故障转移需要的最小 Sentinel 投票数（2 代表 3 台中的 2 台同意即可）
> sentinel monitor mymaster 192.168.0.10 6379 2
>
> # 访问主节点的密码
> sentinel auth-pass mymaster Redis@123
>
> # 主观认为节点宕机的毫秒数（实验环境设为 5000 毫秒 / 5 秒，方便快速看到效果，生产环境一般设 30000）
> sentinel down-after-milliseconds mymaster 5000
>
> # 故障转移的超时时间设为 15 秒
> sentinel failover-timeout mymaster 15000
> EOF
[root@node1 ~]# cat > /usr/lib/systemd/system/redis-sentinel.service << 'EOF'
> [Unit]
> Description=Redis Sentinel
> After=network-online.target
> Wants=network-online.target
>
> [Service]
> Type=simple
> ExecStart=/usr/local/redis/bin/redis-sentinel /etc/redis/redis-sentinel.conf
> ExecStop=/usr/local/redis/bin/redis-cli -p 26379 shutdown
> LimitNOFILE=65535
> Restart=on-failure
> RestartSec=1
> PrivateTmp=true
>
> [Install]
> WantedBy=multi-user.target
> EOF
[root@node1 ~]# systemctl daemon-reload
[root@node1 ~]# systemctl enable redis-sentinel.service --now

```
