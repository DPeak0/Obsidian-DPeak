## 配置MGR
### node1
```bash
[root@node1 ~]# cat > /etc/my.cnf.d/mgr.cnf << EOF
> [mysqld]
> bind-address = 0.0.0.0
> server-id = 1
> log-bin = mysql-bin
> binlog_format = ROW
> # 开启 GTID
> gtid_mode = ON
> enforce_gtid_consistency = ON
> log_slave_updates = ON
> binlog_checksum = NONE
> # MGR 核心组件配置
> plugin_load_add = group_replication.so
> group_replication_group_name = '11111111-2222-3333-4444-555555555555'
> group_replication_start_on_boot = OFF
> group_replication_local_address = '192.168.0.10:33061'
> group_replication_group_seeds = '192.168.0.10:33061,192.168.0.20:33061,192.168.0.30:33061'
> group_replication_single_primary_mode = ON
> transaction_write_set_extraction = XXHASH64
> EOF

[root@node1 ~]# systemctl restart mysqld.service
```

## 创建MGR同步专用账号
### node1
```mysql
[root@node2 ~]# mysql -uroot -p1

mysql> SET SQL_LOG_BIN = 0;
Query OK, 0 rows affected (0.00 sec)

mysql> CREATE USER 'mgr_repl'@'192.168.0.%' IDENTIFIED BY 'Mgr@123456';
Query OK, 0 rows affected (0.02 sec)

mysql> GRANT REPLICATION SLAVE ON *.* TO 'mgr_repl'@'192.168.0.%';
Query OK, 0 rows affected (0.01 sec)

mysql> SET SQL_LOG_BIN = 1;
Query OK, 0 rows affected (0.00 sec)

mysql> CHANGE MASTER TO MASTER_USER='mgr_repl', MASTER_PASSWORD='Mgr@123456' FOR CHANNEL 'group_replication_recovery';
Query OK, 0 rows affected, 2 warnings (0.01 sec)
```

[root@node2 ~]# mysql -uroot -p1 -e "

## 启动MGR集群
### node1
```bash
[root@node1 ~]# mysql -uroot -p1
mysql> SET GLOBAL group_replication_bootstrap_group=ON;
Query OK, 0 rows affected (0.00 sec)

mysql> start group_replication;
Query OK, 0 rows affected (2.18 sec)

mysql> set global group_replication_bootstrap_group=OFF;
Query OK, 0 rows affected (0.00 sec)
```
