## 欢迎词

### 清空/etc/motd文件下的内容

```bash
root@debian:~# > /etc/motd
```

### 编辑欢迎词

```bash
root@debian:~# vim  /etc/update-motd.d/10-uname 
#!/bin/sh
#uname -snrvm
echo "*******************************"
echo "    ChinaSkills 2020 - Wuhan"
echo "\t Module A Linux"
echo "\t  lnxserver1"
echo "\t   dongbowen\n"
echo "\t  >>$(cat /etc/hostname)<<"
echo "\t>>Debian $(cat /etc/debian_version)<<"
echo ">>$(date)<<"
echo "*******************************"
```
