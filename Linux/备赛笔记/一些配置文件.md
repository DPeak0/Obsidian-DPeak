# 配置文件



### HTTP

```bash
ServerRoot "/etc/httpd"			配置文件默认存放目录
Listen 8080						服务监听端口
Include conf.modules.d/*.conf	包涵的配置文件所在目录
User apache						运行服务的用户
Group apache					运行服务的组
ServerAdmin root@localhost		网站管理员邮箱地址
ServerName www.example.com:80	网站的域名设置


<Directory />...</Directory>	目录授权，授权的目录是ServerRoot
    AllowOverride none			不允许权限覆盖		
    Require all denied			拒绝所有访问请求
	
DocumentRoot "/var/www/html"	网站首页存放的默认目录

<IfModule dir_module>
    DirectoryIndex index.html	网站默认首页的文件类型
</IfModule>

```



### DHCP

subnet 192.168.2.0 netmask 255.255.255.0 {
  range 192.168.2.100 192.168.2.200;
  option domain-name-servers 144.144.144.144,8.8.8.8;
  option domain-name "dongbowen.cn";
  option subnet-mask 255.255.255.0;
  option routers 192.168.2.254;
  default-lease-time 86400;
  max-lease-time 259200;
}
host DBW {
hardware ethernet 00:50:56:C0:00:0A;

}
